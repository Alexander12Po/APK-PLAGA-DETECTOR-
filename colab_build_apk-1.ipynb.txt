    "## 6) Compilar el APK\n",
    "\n",
    "Esto puede tardar entre 15 y 40 minutos la primera vez (descarga el\n",
    "Android SDK/NDK 25b completos). Las siguientes compilaciones son mas\n",
    "rapidas porque Buildozer cachea el SDK/NDK en `.buildozer/`.\n",
    "\n",
    "El resultado completo se guarda en `build_log.txt`. Si algo falla, la\n",
    "SIGUIENTE celda te muestra automaticamente solo las lineas con error —\n",
    "no hace falta que busques nada a mano en el log."
   ]
  },
  {
   "cell_type": "code",
   "execution_count": null,
   "metadata": {},
   "outputs": [],
   "source": [
    "%cd /content/{PROJECT_DIR}\n",
    "!rm -rf .buildozer\n",
    "import subprocess, os\n",
    "\n",
    "env = os.environ.copy()\n",
    "with open(\"build_log.txt\", \"w\") as logf:\n",
    "    process = subprocess.Popen(\n",
    "        [\"bash\", \"-lc\", \"source /content/buildenv/bin/activate && yes | buildozer -v android debug\"],\n",
    "        stdout=subprocess.PIPE, stderr=subprocess.STDOUT, env=env, text=True, bufsize=1\n",
    "    )\n",
    "    for line in process.stdout:\n",
    "        print(line, end=\"\")\n",
    "        logf.write(line)\n",
    "    process.wait()\n",
    "    print(\"\\n\\n===== Buildozer termino con codigo:\", process.returncode, \"=====\")"
   ]
